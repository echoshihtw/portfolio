% One-page résumé layout.
%
% The constraint is hard: this must fit on exactly one page, and it must stay
% comfortable to read. Those pull against each other, so the tuning below buys
% space from margins, leading, and section gaps, never from font size, which
% is what actually costs readability.
%
% If it spills to a second page, cut a bullet before shrinking anything here.

\usepackage[left=0.5in,right=0.5in,top=0.34in,bottom=0.26in]{geometry}

\usepackage{xcolor}
\definecolor{ink}{HTML}{1A1A1A}
\definecolor{muted}{HTML}{5B584F}
\definecolor{accent}{HTML}{A07D45}

% Charter runs large for its point size; 0.97 leading keeps lines distinct
% without the airy feel of the default.
\linespread{0.97}

\usepackage{enumitem}
\setlist[itemize]{
  leftmargin=1em,
  labelsep=0.45em,
  itemsep=1.4pt,
  topsep=2pt,
  parsep=0pt,
  partopsep=0pt,
}

% Bullets: a small dot reads quieter than the default bold •, so the eye lands
% on the text rather than the marker.
\renewcommand{\labelitemi}{\color{accent}\raisebox{0.12em}{\tiny$\bullet$}}

\usepackage{titlesec}
\titleformat{\section}
  {\normalsize\bfseries\scshape\color{ink}}{}{0pt}{}
  [\vspace{-1.5pt}{\color{accent}\titlerule[0.6pt]}]
% Company is the anchor when scanning. Distinguished by weight and colour
% rather than size: a larger company name cost four lines across four jobs
% and pushed the résumé onto a second page.
\titleformat{\subsection}{\normalsize\bfseries\color{ink}}{}{0pt}{}

% Asymmetric spacing: a clear gap above each section, a tight one below, so
% headings group with their own content instead of floating between blocks.
\titlespacing*{\section}{0pt}{6pt}{3pt}
\titlespacing*{\subsection}{0pt}{6pt}{2pt}

% A job's heading, two lines that scan like a table without being one: the
% company bold at the left with the dates at the right margin, then the role
% with the location after it, both quieter. scripts/resume-entry.lua emits
% this in place of the "## Company" heading and the line under it, for the
% PDF only. \hbox to \linewidth rather than \hfill in a paragraph: under
% \RaggedRight the last line's fill would share the stretch and the dates
% would stop short of the margin. Arguments: company, dates, role, location.
% \parskip is zeroed inside, or each of the three \par would add its own
% 1.2pt on top of the spacing set here, which is what pushed Education off
% the page the first time.
\newcommand{\cventry}[4]{%
  \par\begingroup\setlength{\parskip}{0pt}%
  \vspace{5pt}%
  \noindent\hbox to \linewidth{%
    {\normalsize\bfseries\color{ink}#1}\hfil{\small\color{muted}#2}}%
  \par\vspace{1pt}%
  \noindent{\normalsize\color{ink}#3}%
  \ifx\relax#4\relax\else{\small\color{muted}\ \textperiodcentered\ #4}\fi%
  \par\endgroup%
}

\setlength{\parskip}{1.2pt}
\setlength{\parindent}{0pt}
\setlength{\emergencystretch}{2em}

% ATS-friendly: no hyphenation and ragged-right, so no word is split across
% lines and résumé scanners extract clean text.
\usepackage[none]{hyphenat}
\usepackage{ragged2e}
\RaggedRight

% Widow/orphan control: a lone bullet stranded at a page or column edge is the
% most common way a tight résumé starts looking sloppy.
\widowpenalty=10000
\clubpenalty=10000
